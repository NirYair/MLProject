קוד: טעינת הנתונים

from google.colab import drive
import os
import cv2
import numpy as np
import matplotlib.pyplot as plt

# Mount Google Drive
drive.mount('/content/drive')

# Base paths for train and test directories
base_dir = '/content/drive/MyDrive/ML_Project/Rock_Paper_Scissors/Rock-Paper-Scissors'
train_dir = os.path.join(base_dir, 'train')
test_dir = os.path.join(base_dir, 'test')

categories = ['rock', 'paper', 'scissors']
IMG_SIZE = 64 # Resizing to keep execution fast and prevent memory issues

def load_data(directory):
    """Load images and labels directly from the specified directory."""
    X, y = [], []
    for category in categories:
        folder_path = os.path.join(directory, category)
        if not os.path.exists(folder_path):
            continue
            
        label = categories.index(category)
        for img_name in os.listdir(folder_path):
            img_path = os.path.join(folder_path, img_name)
            # Load original RGB image for presentation purposes
            img = cv2.imread(img_path)
            if img is not None:
                X.append(cv2.cvtColor(img, cv2.COLOR_BGR2RGB))
                y.append(label)
                
    # Shuffle the dataset
    indices = np.arange(len(X))
    np.random.shuffle(indices)
    return np.array([X[i] for i in indices]), np.array([y[i] for i in indices])

print("Loading Train and Test datasets separately...")
X_train_raw, y_train = load_data(train_dir)
X_test_raw, y_test = load_data(test_dir)

print(f"Loaded {len(X_train_raw)} Train images and {len(X_test_raw)} Test images.")

def plot_first_5(images, labels, title):
    """Visual presentation of the first 5 rows in the dataset."""
    if len(images) == 0: return
    plt.figure(figsize=(12, 3))
    for i in range(min(5, len(images))):
        plt.subplot(1, 5, i+1)
        plt.imshow(images[i])
        plt.title(categories[labels[i]].capitalize())
        plt.axis('off')
    plt.suptitle(title, fontsize=14)
    plt.show()

# Show the first 5 images of both trainset and testset
plot_first_5(X_train_raw, y_train, "Part 1: First 5 Images of Train Dataset")
plot_first_5(X_test_raw, y_test, "Part 1: First 5 Images of Test Dataset")

חלק 2 – Feature Engineering

מכיוון שאנו משתמשים באלגוריתם KNN (המחשב מרחק מתמטי בין נקודות), לא ניתן להזין תמונה צבעונית דו-מימדית כפי שהיא. תהליך המרת התכונות כולל:
1. המרת צבע: הפיכת התמונה ל-Grayscale (שחור-לבן) כדי להסיר מידע לא רלוונטי ולהתרכז בצורת היד.
2. השטחה (Flattening): הפיכת התמונה הדו-מימדית לווקטור (מערך) חד-מימדי.
3. נרמול (Scaling): שימוש ב-StandardScaler כדי לאזן את ערכי הפיקסלים, צעד קריטי לאלגוריתמים מבוססי מרחק. 

from sklearn.preprocessing import StandardScaler

def apply_feature_engineering(img_rgb):
    """Convert image to grayscale, resize, and flatten to a 1D vector."""
    # Convert to grayscale
    img_gray = cv2.cvtColor(img_rgb, cv2.COLOR_RGB2GRAY)
    # Resize
    img_resized = cv2.resize(img_gray, (IMG_SIZE, IMG_SIZE))
    # Flatten
    return img_resized.flatten()

# Apply feature engineering to the raw datasets
X_train_features = np.array([apply_feature_engineering(img) for img in X_train_raw])
X_test_features = np.array([apply_feature_engineering(img) for img in X_test_raw])

# Normalize the data using StandardScaler
scaler = StandardScaler()
X_train_scaled = scaler.fit_transform(X_train_features)
X_test_scaled = scaler.transform(X_test_features)

# Present 2-3 examples of the Feature Engineering process
print("\n--- Feature Engineering Process: 3 Examples from Train Set ---")
for i in range(3):
    original_shape = X_train_raw[i].shape
    final_shape = X_train_scaled[i].shape
    print(f"Example {i+1}: Original RGB Shape {original_shape} --> Flattened & Scaled Shape {final_shape}")

חלק 3 ו-4 – מימוש אלגוריתם הלמידה ואימון

האלגוריתם שנבחר הוא KNN (K-Nearest Neighbors). הגדרנו Hyperparameters כדי לשפר את הביצועים: n_neighbors=5 (בדיקת 5 השכנים הקרובים) ו-weights='distance' (מתן משקל רב יותר לשכנים קרובים). לאחר מכן, הפעלנו את האימון על ה-Trainset המעובד. 

from sklearn.neighbors import KNeighborsClassifier

print("--- Part 3 & 4: Algorithm Implementation & Training ---")

# Define hyperparameters
hyperparameters = {
    'n_neighbors': 5,
    'weights': 'distance', 
    'metric': 'euclidean'
}

print(f"Initializing KNN with hyperparameters: {hyperparameters}")
knn_model = KNeighborsClassifier(**hyperparameters)

# Execute the training flow on the entire trainset
print("Training the model on the full Trainset using fit()...")
knn_model.fit(X_train_scaled, y_train)
print("Training Completed Successfully.")

חלק 5 – חיזוי ושערוך איכות המודל

בשלב זה המודל חוזה את התוויות עבור ה-Testset. מכיוון שמדובר בסיווג מרובה מחלקות (Multi-class) ללא מחלקה מרכזית אחת, נשתמש במדד Macro Average F1-Score לפי דרישות המטלה כדי לשערך את איכות המודל. 

from sklearn.metrics import f1_score

print("--- Part 5: Prediction and Evaluation on Test Set ---")

if len(X_test_scaled) > 0:
    # Predict the classifications on the test set
    y_pred = knn_model.predict(X_test_scaled)
    
    # Visualize the first 5 predictions on the test set
    plt.figure(figsize=(12, 4))
    for i in range(min(5, len(X_test_raw))):
        plt.subplot(1, 5, i+1)
        plt.imshow(X_test_raw[i]) 
        
        actual_label = categories[y_test[i]].capitalize()
        pred_label = categories[y_pred[i]].capitalize()
        
        text_color = 'green' if y_test[i] == y_pred[i] else 'red'
        plt.title(f"Actual: {actual_label}\nPred: {pred_label}", color=text_color, fontweight='bold')
        plt.axis('off')
        
    plt.suptitle("First 5 Predictions vs Actual (Test Set)", fontsize=16)
    plt.tight_layout()
    plt.show()
    
    # Calculate Macro Average F1-Score
    macro_f1 = f1_score(y_test, y_pred, average='macro')
    print(f"\nModel Quality (Macro Average F1-Score): {macro_f1:.4f}")
else:
    print("Test set is empty. Cannot perform evaluation.")
